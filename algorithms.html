📋 Algorithm Information
What it does Scans each element one by one from start to end until the target is found or the list is exhausted. Repeatedly halves a sorted array to locate the target by comparing with the midpoint. Repeatedly swaps adjacent elements that are out of order, "bubbling" the largest to the end each pass. Divides the array into halves recursively, then merges them back in sorted order. Builds a sorted portion one element at a time by inserting each new element into its correct position.
Pseudo code
linearSearch(arr, target): n = length(arr) for i = 0 to n - 1: if arr[i] == target: return i return -1
BinarySearch(arr, target): lo = 0 hi = length(arr) - 1 while lo <= hi: mid = (lo + hi) // 2 if arr[mid] == target: return mid elif arr[mid] < target: lo = mid + 1 else: hi = mid - 1 return -1
BubbleSort(arr): n = length(arr) for i = 0 to n - 2: swapped = false for j = 0 to n - i - 2: if arr[j] > arr[j + 1]: swap(arr[j], arr[j + 1]) swapped = true if not swapped: break return arr
def merge_sort(arr): if len(arr) <= 1: return arr mid = len(arr) // 2 LA = merge_sort(arr[:mid]) RA = merge_sort(arr[mid:]) return merge(LA, RA) def merge(LA, RA): result = [] i = 0 j = 0 while i < len(LA) and j < len(RA): if LA[i] <= RA[j]: result.append(LA[i]) i = i + 1 else: result.append(RA[j]) j = j + 1 while i < len(LA): result.append(LA[i]) i = i + 1 while j < len(RA): result.append(RA[j]) j = j + 1 return result
insertionSort(arr): n = length(arr) for i = 1 to n - 1: key = arr[i] j = i - 1 while j >= 0 and arr[j] > key: arr[j + 1] = arr[j] j = j - 1 arr[j + 1] = key return arr
